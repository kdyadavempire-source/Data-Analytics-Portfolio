SALES & REVENUE ANALYTICS
Power BI DAX Measures

--------------------------------------------------
1. Total Revenue
--------------------------------------------------

Total Revenue =
SUM(sales_transactions[Sales])


--------------------------------------------------
2. Total Cost
--------------------------------------------------

Total Cost =
SUM(sales_transactions[Cost])


--------------------------------------------------
3. Total Profit
--------------------------------------------------

Total Profit =
SUM(sales_transactions[Profit])


--------------------------------------------------
4. Total Orders
--------------------------------------------------

Total Orders =
DISTINCTCOUNT(sales_transactions[Order ID])


--------------------------------------------------
5. Total Quantity
--------------------------------------------------

Total Quantity =
SUM(sales_transactions[Quantity])


--------------------------------------------------
6. Average Order Value
--------------------------------------------------

Average Order Value =
DIVIDE(
    [Total Revenue],
    [Total Orders]
)


--------------------------------------------------
7. Profit Margin
--------------------------------------------------

Profit Margin =
DIVIDE(
    [Total Profit],
    [Total Revenue],
    0
)


--------------------------------------------------
8. Revenue Growth
--------------------------------------------------

Revenue Growth =
VAR CurrentRevenue = [Total Revenue]
VAR PreviousRevenue =
    CALCULATE(
        [Total Revenue],
        DATEADD(
            sales_transactions[Order Date],
            -1,
            YEAR
        )
    )
RETURN
DIVIDE(
    CurrentRevenue - PreviousRevenue,
    PreviousRevenue,
    0
)
